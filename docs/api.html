javascript


const base = "https://kaungkhantko26.github.io/space-mission-dataset/api/v1";
const get = p => fetch(`${base}/${p}.json`).then(r => r.json());
const mars = await get("planets/mars");
const {records} = await get("launch_vehicle_configs");
const heavy = records.filter(r => r.active && r.leo_capacity_kg > 15000);


python


import json, urllib.request
base = "https://kaungkhantko26.github.io/space-mission-dataset/api/v1"
get = lambda p: json.load(urllib.request.urlopen(f"{base}/{p}.json"))
mars = get("planets/mars")
print(mars["gravitational_parameter_km3_s2"], mars["data_class"])

conventions


    

  • SI units unless the field name says otherwise (_km, _kg, _kn, _t, _days).
    • 

  • Absent field = not verified. Never zero-filled.
    • 

  • Every record: data_class, source_ids, and field_verification or collection provenance. Aggregator and vendor data carry source_authority.
    • 

  • Ids are unique within a collection. v1 may gain fields but existing fields keep their meaning.
    •
